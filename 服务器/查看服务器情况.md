``` bash
echo "=== 发行版 ==="
grep -E '^(PRETTY_NAME|VERSION_ID|VERSION_CODENAME)=' /etc/os-release

echo "=== 内核 ==="
uname -r

echo "=== 架构 ==="
uname -m
```


``` bash
echo "=== 当前在线 ==="
who
echo "当前会话数: $(who | wc -l)"
echo "当前用户数: $(who | awk '{print $1}' | sort -u | wc -l)"
echo

echo "=== 最近登录 ==="
last -n 20
echo

echo "=== 每个用户最后登录 ==="
lastlog 2>/dev/null | grep -v 'Never logged in'
echo

echo "=== 最近重启 ==="
last reboot -n 5
echo

echo "=== 负载/运行时长 ==="
uptime
```
